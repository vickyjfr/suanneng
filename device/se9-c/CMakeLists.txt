cmake_minimum_required(VERSION 3.16)
project(se9_device LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SE9_WEBRTC "Build the NAS WebRTC transport" ON)
option(SE9_USE_LIBNICE "Enable TURN UDP/TCP/TLS through libnice" ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
include(FetchContent)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
set(ENABLE_CJSON_TEST OFF CACHE BOOL "" FORCE)
set(ENABLE_CJSON_UTILS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SE9_THIRD_PARTY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party" CACHE PATH "Local third-party sources; missing ones are fetched")
if(NOT DEFINED FETCHCONTENT_SOURCE_DIR_CJSON AND EXISTS "${SE9_THIRD_PARTY_DIR}/cJSON/CMakeLists.txt")
 set(FETCHCONTENT_SOURCE_DIR_CJSON "${SE9_THIRD_PARTY_DIR}/cJSON")
endif()
FetchContent_Declare(cjson GIT_REPOSITORY https://github.com/DaveGamble/cJSON.git GIT_TAG c859b25da02955fef659d658b8f324b5cde87be3)
FetchContent_MakeAvailable(cjson)
set(SE9_CLOUD_GRANT_PUBLIC_KEY_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../deploy/cloud-grant-public.pem" CACHE FILEPATH "Cloud authorization public key to embed")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SE9_CLOUD_GRANT_PUBLIC_KEY_FILE}")
file(READ "${SE9_CLOUD_GRANT_PUBLIC_KEY_FILE}" SE9_CLOUD_GRANT_PEM)
string(REPLACE "\n" "\\n" SE9_CLOUD_GRANT_PEM_ESCAPED "${SE9_CLOUD_GRANT_PEM}")
configure_file(rtc/cloud_grant_key.h.in generated/se9_cloud_grant_key.h @ONLY)
add_library(se9_common STATIC rtc/ft_common.c rtc/device_config.c)
target_include_directories(se9_common PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/rtc ${cjson_SOURCE_DIR})
target_include_directories(se9_common PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(se9_common PUBLIC cjson OpenSSL::Crypto Threads::Threads)
add_executable(se9_file_api se9_file_api.c rtc/transfer_store.c)
target_compile_options(se9_file_api PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(se9_file_api PRIVATE se9_common Threads::Threads)
add_executable(se9_agent se9_agent.c)
target_compile_options(se9_agent PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(se9_agent PRIVATE se9_common OpenSSL::SSL Threads::Threads)
if(SE9_WEBRTC)
 set(USE_NICE ${SE9_USE_LIBNICE} CACHE BOOL "" FORCE)
 set(NO_MEDIA ON CACHE BOOL "" FORCE)
 set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
 set(NO_TESTS ON CACHE BOOL "" FORCE)
 set(RTC_UPDATE_VERSION_HEADER OFF CACHE BOOL "" FORCE)
 if(NOT DEFINED FETCHCONTENT_SOURCE_DIR_DATACHANNEL AND EXISTS "${SE9_THIRD_PARTY_DIR}/libdatachannel/deps/usrsctp/CMakeLists.txt")
  set(FETCHCONTENT_SOURCE_DIR_DATACHANNEL "${SE9_THIRD_PARTY_DIR}/libdatachannel")
 endif()
 FetchContent_Declare(datachannel GIT_REPOSITORY https://github.com/paullouisageneau/libdatachannel.git GIT_TAG 6b1e2e620f1e37f0eafeee702eaea0043cb305fd)
 FetchContent_MakeAvailable(datachannel)
 add_executable(se9_rtc_probe tests/rtc_probe.c)
 target_link_libraries(se9_rtc_probe PRIVATE se9_common datachannel-static)
 set_target_properties(se9_rtc_probe PROPERTIES LINKER_LANGUAGE CXX)
 target_sources(se9_agent PRIVATE rtc/rtc_transport.c rtc/rtc_socket.c)
 target_compile_definitions(se9_agent PRIVATE SE9_WITH_WEBRTC=1)
 target_link_libraries(se9_agent PRIVATE datachannel-static)
 set_target_properties(se9_agent PROPERTIES LINKER_LANGUAGE CXX)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(se9_config_test rtc/device_config_test.c)
 target_link_libraries(se9_config_test PRIVATE se9_common)
 target_compile_options(se9_config_test PRIVATE -UNDEBUG)
 add_test(NAME se9_device_config COMMAND se9_config_test)
 add_executable(se9_protocol_test rtc/ft_common_test.c)
 target_link_libraries(se9_protocol_test PRIVATE se9_common)
 target_compile_options(se9_protocol_test PRIVATE -UNDEBUG)
 add_test(NAME se9_protocol COMMAND se9_protocol_test)
 find_package(Python3 COMPONENTS Interpreter QUIET)
 if(Python3_Interpreter_FOUND)
  add_test(NAME se9_installer COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/install_nas_test.py)
  set_tests_properties(se9_installer PROPERTIES ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
  add_test(NAME se9_ip_certificate COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/ip_certificate_check.py $<TARGET_FILE:se9_agent>)
  set_tests_properties(se9_ip_certificate PROPERTIES RUN_SERIAL TRUE)
  add_test(NAME se9_transfer_store COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/transfer_store_check.py $<TARGET_FILE:se9_file_api>)
 endif()
endif()
